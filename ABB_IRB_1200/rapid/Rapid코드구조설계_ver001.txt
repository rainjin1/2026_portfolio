ex
베이스 패널 적재함에 3개가 있을때 들어가야 하는 높이 = 베이스패널 적재함에 1개 들어있을때 좌표의 z값 + (베이스패널수량 -1)*베이스패널 전체 높이

WaitTime 0.5;
endproc

proc 흡착오프();

WaitTime 0.5;
endproc

proc wall to con();

	move j offs(하양벽패널적재함에 1개 있는 좌표, 0, 0, 하양벽패널적재함에 1개있는좌표의 z값 + {(하양벽패널수량 -1)*벽패널굵기} + 벽패널적재함 지지대 높이 + 50), v200, z50, tool1;
	move l offs(하양벽패널적재함에 1개 있는 좌표, 0, 0, 하양벽패널적재함에 1개있는좌표의 z값 + {(하양벽패널수량 -1)*벽패널굵기}), v50, fine, tool1;

	move l offs(하양벽패널적재함에 1개 있는 좌표, 0, 0, 하양벽패널적재함에 1개있는좌표의 z값 + {(하양벽패널수량 -1)*벽패널굵기} + 벽패널적재함 지지대 높이 + 50), v50, z50, tool1;
	move j offs(컨베이어에 베이스 패널 1개 있을때 좌표, 0, 0, 50), v200, z50, tool1;
